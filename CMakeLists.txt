cmake_minimum_required(VERSION 3.25)
project(cpp_opp_projects VERSION 0.1.0 LANGUAGES CXX)
set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
add_executable(cpp_app src/main.cpp)
target_compile_features(cpp_app PRIVATE cxx_std_20)
set_target_properties(cpp_app PROPERTIES CXX_EXTENSIONS OFF)
if(MSVC)
  target_compile_options(cpp_app PRIVATE /W4 /permissive- /EHsc)
else()
  target_compile_options(cpp_app PRIVATE -Wall -Wextra -Wpedantic)
endif()
enable_testing()
add_test(NAME smoke COMMAND cpp_app)
set_tests_properties(smoke PROPERTIES PASS_REGULAR_EXPRESSION "C\\+\\+ is ready")
install(TARGETS cpp_app RUNTIME DESTINATION bin)
set(CPACK_GENERATOR ZIP)
set(CPACK_PACKAGE_DIRECTORY "${CMAKE_SOURCE_DIR}/dist")
include(CPack)
